import 'package:bloom/bloom.dart';
import 'package:bloom_ui/bloom_ui.dart';

@Route('/dashboard')
class DashboardPage extends BloomPage {
  // ⚡️ Zero-boilerplate reactive signal
  final revenue = signal(24500);
  late final formatted = computed(() => '\$${revenue.value}');

  @override
  Widget build(BuildContext context) {
    return BloomScaffold(
      body: Center(
        child: Column(
          children: [
            BloomMetricCard(
              title: 'Monthly Recurring Revenue',
              value: formatted.value, // Auto-rebuilds!
              trend: '+14%',
            ),
            BloomButton(
              onPressed: () => revenue.value += 1200,
              label: 'Simulate Sale',
            ),
          ],
        ),
      ),
    );
  }
}

import 'package:bloom/bloom.dart';

@RoutePage()
class ProductDetails extends BloomPage {
  // Automatically mapped from URL
  @PathParam('id')
  final String productId;

  ProductDetails({required this.productId});

  @override
  Widget build(BuildContext context) {
    final product = useQuery(fetchProduct(productId));

    return product.when(
      data: (data) => Text(data.name),
      loading: () => BloomSkeleton(),
      error: (e) => Text('Failed to load'),
    );
  }
}
